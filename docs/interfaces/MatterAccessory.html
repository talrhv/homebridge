    Interface MatterAccessory<T>

    Matter Accessory - Plugin API Interface

    This is the main interface that plugin developers use to register Matter accessories with Homebridge.

    interface MatterAccessory<T = Record<string, any>> {
        clusters: {
            [clusterName: string]: { [attributeName: string]: unknown };
        };
        context?: T;
        deviceType: EndpointType;
        displayName: string;
        firmwareRevision?: string;
        getState?: (cluster: string, attribute: string) => any;
        handlers?: { [clusterName: string]: MatterClusterHandlers };
        hardwareRevision?: string;
        manufacturer: string;
        model: string;
        serialNumber: string;
        softwareVersion?: string;
        uuid: string;
    }
